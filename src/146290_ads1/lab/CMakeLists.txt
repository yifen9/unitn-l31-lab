cmake_minimum_required(VERSION 3.20)
project(ads_labs C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()

add_library(common_headers INTERFACE)
target_include_directories(common_headers INTERFACE ${CMAKE_SOURCE_DIR}/include)

file(GLOB LABS RELATIVE ${CMAKE_SOURCE_DIR}/src ${CMAKE_SOURCE_DIR}/src/*)
foreach(L ${LABS})
  if(IS_DIRECTORY ${CMAKE_SOURCE_DIR}/src/${L})
    set(LAB_DIR ${CMAKE_SOURCE_DIR}/src/${L})
    file(GLOB EXES RELATIVE ${LAB_DIR}/src ${LAB_DIR}/src/*)
    set(RUN_DEPS)
    foreach(E ${EXES})
      if(IS_DIRECTORY ${LAB_DIR}/src/${E})
        set(EXE_NAME ${L}_${E})
        file(GLOB EX_SRCS CONFIGURE_DEPENDS ${LAB_DIR}/src/${E}/*.c)
        add_executable(${EXE_NAME} ${EX_SRCS})
        target_include_directories(${EXE_NAME} PRIVATE ${LAB_DIR}/include ${CMAKE_SOURCE_DIR}/include)
        target_link_libraries(${EXE_NAME} PRIVATE common_headers)
        add_custom_target(run_${L}_${E} COMMAND $<TARGET_FILE:${EXE_NAME}> DEPENDS ${EXE_NAME} USES_TERMINAL)
        list(APPEND RUN_DEPS run_${L}_${E})
        file(GLOB CASES RELATIVE ${LAB_DIR}/tests/${E} ${LAB_DIR}/tests/${E}/*)
        foreach(CASE ${CASES})
          if(IS_DIRECTORY ${LAB_DIR}/tests/${E}/${CASE})
            set(IN ${LAB_DIR}/tests/${E}/${CASE}/i.txt)
            set(OUT ${LAB_DIR}/tests/${E}/${CASE}/o.txt)
            if(EXISTS ${IN} AND EXISTS ${OUT})
              add_test(NAME ${L}_${E}_${CASE}
                       COMMAND ${CMAKE_COMMAND}
                               -Dexe=$<TARGET_FILE:${EXE_NAME}>
                               -Dinput=${IN}
                               -Dexpected=${OUT}
                               -P ${CMAKE_SOURCE_DIR}/scripts/RunIO.cmake)
            endif()
          endif()
        endforeach()
      endif()
    endforeach()
    if(RUN_DEPS)
      add_custom_target(run_${L}_all DEPENDS ${RUN_DEPS})
    endif()
  endif()
endforeach()
